function [jerk_star,a,v,theta]= fun(theta_d,theta_last,v_last,a_last,jerk_last)
 %% 设置各项参数
             coder.extrinsic('quadprog');
             coder.extrinsic('full');
              coder.extrinsic('optimoptions');
             theta_up = [pi/2 pi/2 pi]';
             theta_low = -[pi/2 pi/2 pi]';
%              v_up = [3 3 3]';
%              v_low = -[3 3 3]';
%              a_up = [8 8 8]';
%              a_low = -[8 8 8]';
%              jerk_up = [30 30 30]';
%              jerk_low = -[30 30 30]';
             v_up = [1 1 1]';
             v_low = -[1 1 1]';
             a_up = [3 3 3]';
             a_low = -[3 3 3]';
             jerk_up = [12 12 12]';
             jerk_low = -[12 12 12]';

             xi = 10;
             lambda_p = 5;
             lambda_a = 5;
             lambda_v = 5;
             Ts = 0.02;
             xub = zeros(3,1);
             xlb = zeros(3,1);
             for i=1:1:3
                 xub(i) = min([-3*lambda_p*a_last(i)-3*lambda_p^2*v_last(i)+lambda_p^3*(theta_up(i)-theta_last(i)) -2*lambda_v*a_last(i)+lambda_v^2*(v_up(i)-v_last(i)) lambda_a*(a_up(i)-a_last(i)) jerk_up(i)]);
                 xlb(i) = max([-3*lambda_p*a_last(i)-3*lambda_p^2*v_last(i)-lambda_p^3*(theta_last(i)-theta_low(i)) -2*lambda_v*a_last(i)+lambda_v^2*(v_low(i)-v_last(i)) lambda_a*(a_low(i)-a_last(i)) jerk_low(i)]);
             end
             pk = 3*xi*a_last+3*xi^2*v_last+xi^3*(theta_last-theta_d);
             H = eye(3,3);
             f = pk;
             A = eye(3,3);
             b = xub;
             Aeq = zeros(3,3);
             beq = zeros(3,1);
             lb = xlb;
             x0 = jerk_last;
             % 求解二次规划问题 min 1/2*xHx+f'x ,s,t:Ax<=b,lb<=x<=ub
             % options = optimoptions('quadprog','Algorithm','interior-point-convex');%interior-point-convex/trust-region-reflective
             x = zeros(3,1);
             opts = optimoptions('quadprog','Algorithm','interior-point-convex');%interior-point-convex/trust-region-reflective
             [x,fval,exitflag,output,lambda] =quadprog(H,f,A,b,Aeq,beq,lb,xub,x0,opts); %求取最小值时x值
             jerk_star = x;
             a = a_last + jerk_star*Ts;
             v = v_last+a*Ts;
             theta = theta_last+v*Ts;

end
